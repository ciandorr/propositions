\section{Bullet symbols by nesting depth}

Standard \verb|\itemize| uses different symbols at each level.
The \texttt{bullet} type now does the same via \verb|\proplevelchoice|.

\begin{prop}
  \bitem Level 1: should be \textbullet
  \begin{prop}
    \bitem Level 2: should be --
    \begin{prop}
      \bitem Level 3: should be \textasteriskcentered
      \begin{prop}
        \bitem Level 4: should be \textperiodcentered
        \begin{prop}
          \bitem Level 5+: should still be \textperiodcentered{} (last in list)
        \end{prop}
      \end{prop}
    \end{prop}
  \end{prop}
  \bitem Back to level 1: \textbullet{} again
\end{prop}

\section{Custom symbols via \texttt{\textbackslash proplevelchoice}}

\begin{prop}
  \pitem[name=\proplevelchoice{A,B,C}] Level 1: should be A
  \begin{prop}
    \pitem[name=\proplevelchoice{A,B,C}] Level 2: should be B
    \begin{prop}
      \pitem[name=\proplevelchoice{A,B,C}] Level 3: should be C
      \begin{prop}
        \pitem[name=\proplevelchoice{A,B,C}] Level 4: should still be C
      \end{prop}
    \end{prop}
  \end{prop}
\end{prop}
