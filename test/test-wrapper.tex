%% The "framed" style is built into the package (available when mdframed or
%% tcolorbox is loaded): it frames the list, keeping the level's effective left
%% margin and centring the frame one padding to the left of the contents.  It is
%% used directly below, and inherited by framedbold.

%% A custom coloured variant, demonstrating a user-defined wrapper with the same
%% geometry as the built-in framed style plus a tinted background.
\newlength\framedpad    \setlength\framedpad{6pt}
\newlength\framedouter  \setlength\framedouter{\dimexpr\leftmargini-\framedpad\relax}
\SetPropStyle{shaded}{
  align = flush, leftmargin = 0pt, rightmargin = 0pt,
  wrapper begin = {\begin{mdframed}[
      linewidth=0.4pt, linecolor=blue!40, backgroundcolor=blue!8,
      leftmargin=\framedouter, rightmargin=\framedouter,
      innerleftmargin=\framedpad, innerrightmargin=\framedpad,
      innertopmargin=\framedpad, innerbottommargin=\framedpad,
      skipabove=\topsep, skipbelow=\topsep]},
  wrapper end   = {\end{mdframed}},
}

%% Inheritance: a style whose parent is the built-in "framed" inherits the frame
%% and adds item-level formatting (bold names).
\SetPropStyle{framedbold}{
  style  = framed,
  format = \textbf{#1},
}

\section{Geometry: framed vs.\ unframed}

The horizontal rules mark the surrounding text margins.  The framed list keeps
the same effective left margin as the unframed body: the contents sit where the
unframed body would, one padding to the right of the frame rule, and the frame
is centred so the right side is symmetric.  Framed items are flush, so their
labels sit at the content edge rather than hanging into the padding.

\noindent\rule{\textwidth}{0.3pt}

\begin{prop}
  \item[name=Claim] A named unframed proposition, long enough to wrap onto a
  second line so the left and right body edges are visible for comparison.
  \item[counter=none] An unlabelled unframed proposition.
\end{prop}

\begin{prop}[style=framed]
  \item[name=Claim] A named framed proposition, long enough to wrap onto a second
  line so the left and right body edges are visible for comparison.
  \item[counter=none] An unlabelled framed proposition, long enough to wrap onto a second
  line so the left and right body edges are visible for comparison.
  \item A numbered framed proposition, long enough to wrap onto a second
  line so the left and right body edges are visible for comparison.  (We hope users won't want to do this!)
\end{prop}

\section{Nesting: only the outermost list is framed}

Only the outer list should get a frame; the sublist sits inside that same
frame and is not separately framed.

\begin{prop}[style=framed]
  \item Outer item, inside the frame. \label{w:outer}
  \begin{prop}
    \item Inner item (sublist is not separately framed).
    \item Second inner item.
  \end{prop}
  \item Back to the outer level. \label{w:back}
\end{prop}

References: \ref{w:outer}, \ref{w:back}.

A standard \texttt{itemize} nested in a framed prop keeps its bullets and is
not framed:

\begin{prop}[style=framed]
  \item A proposition.
  \begin{itemize}
    \item an ordinary bullet
    \item another bullet
  \end{itemize}
  \item Another proposition.
\end{prop}

\section{\texttt{\textbackslash propoptions\{style=framed\}} as a scoped default}

Both lists in this group are framed by default:

\begin{prop}
  \item Framed by the group-level default.
\end{prop}

\begin{prop}
  \item Also framed by the same default.
\end{prop}
\endgroup

Outside the group, the default is gone, so this list is unframed:

\begin{prop}
  \item Not framed.
\end{prop}

\section{Nesting wrappers explicitly}

A sublist that requests a wrapper of its own is framed separately, inside the
outer frame.  A \emph{different} wrapper (plain frame outside, shaded inside):

\begin{prop}[style=framed]
  \item Outer item. \label{w:no}
  \begin{prop}[style=shaded]
    \item Inner, shaded.
    \item Inner, shaded two.
  \end{prop}
  \item Outer item again.
\end{prop}

The \emph{same} wrapper explicitly re-requested also nests (frame inside a
frame); only an \emph{inherited}, unrequested wrapper is suppressed:

\begin{prop}[style=framed]
  \item Outer item.
  \begin{prop}[style=framed]
    \item Inner, its own frame.
  \end{prop}
  \item Outer item again.
\end{prop}

\section{\texttt{\textbackslash propoptions} inside a framed environment}

\texttt{\textbackslash propoptions\{style=shaded\}} issued inside the outer
(framed) environment---no explicit group needed---makes each following sublist
shaded; the outer frame is unaffected and still closes correctly:

\begin{prop}[style=framed]
  \item Outer, before switching the sublist default.
  \propoptions{style=shaded}
  \begin{prop}
    \item First shaded sublist.
  \end{prop}
  \begin{prop}
    \item Second shaded sublist.
  \end{prop}
  \item Outer, after (still only the outer frame).
\end{prop}
